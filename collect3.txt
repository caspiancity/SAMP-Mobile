CVehicleModelInfo::CVehicleStructure::CVehicleStructure() : m_aUpgrades()
{
    for (auto& vecPos : m_avDummyPos)
        vecPos.Set(0.0F, 0.0F, 0.0F);

    for (auto& upgrade : m_aUpgrades)
        upgrade.m_nParentComponentId = -1;

    memset(m_apExtras, 0, sizeof(m_apExtras));
    m_nNumExtras = 0;
    m_nMaskComponentsDamagable = 0;
}

CVehicleModelInfo::CVehicleStructure::~CVehicleStructure()
{
    for (int32 i = 0; i < m_nNumExtras; ++i) {
        auto atomic = m_apExtras[i];
        auto frame = RpAtomicGetFrame(atomic);
        RpAtomicDestroy(atomic);
        RwFrameDestroy(frame);
    }
}

    auto pVeh = (CVehicle*)data;

    if(mat->texture) {

        }

            return mat;
        }

    if ( color == 0xff00ff3c ) { // first color 60, 255, 0

            mat->texture = pVeh->m_pVinylTex;
            mat->color = {255, 255, 255, 255};
            return mat;
        }
        mat->color = pVeh->mainColor;
        return mat;
    }
    if ( color == 0xffaf00ff ) { // second color 255, 0, 175
        AddStoredMaterial(RpMaterialGetColor(mat));
        mat->color = pVeh->secondColor;
        return mat;
    }

void ScrSetVehicleNumberPlate(RPCParameters* rpcParams)
{
	unsigned char* Data = reinterpret_cast<unsigned char*>(rpcParams->input);
	int iBitLength = rpcParams->numberOfBitsOfData;

	VEHICLEID VehicleID;
	uint8_t byteLength;
	char szPlateName[32+1];
	RakNet::BitStream bsData(Data, (iBitLength / 8) + 1, false);
	bsData.Read(VehicleID);
	bsData.Read(byteLength);
	if (byteLength > 32) return;

	bsData.Read(szPlateName, byteLength);
	szPlateName[byteLength] = '\0';

	CVehiclePool* pVehiclePool = pNetGame->GetVehiclePool();
	if (!pVehiclePool) return;

	CVehicle* pVehicle = pVehiclePool->GetAt(VehicleID);
	if (!pVehicle) return;

	pVehicle->SetPlateText(szPlateName);
}

    RwCameraSetNearClipPlane(m_camera, 0.01f);
    RwV2d view = { 0.5f, 0.5f };
    RwCameraSetViewWindow(m_camera, &view);
    RwCameraSetProjection(m_camera, rwPERSPECTIVE);
    RpWorldAddCamera(Scene.m_pRpWorld, m_camera);

void MaterialTextGenerator::Render(const char* text, const ImVec2& size, int font_size, bool bold,
								   const ImColor& font_color, const ImColor& background_color, int text_align)
{
	if (pUI) {
		ImGui::NewFrame();

		auto info = Encoding::cp2utf(text).c_str();

		pUI->renderer()->drawRect(ImVec2(0.0f, 0.0f), size, background_color, true);
		ImVec2 text_size = pUI->renderer()->calculateTextSize(info, font_size);

		ImVec2 text_pos;
		switch (text_align)
		{
			case 0: // LEFT
				text_pos = ImVec2(0.0f, 0.0f);
				break;
			case 1: // CENTER
				text_pos = ImVec2((size.x - text_size.x) / 2, (size.y - text_size.y) / 2);
				break;
			case 2: // RIGHT
				text_pos = ImVec2(size.x - text_size.x, 0.0f);
				break;
		}

		pUI->renderer()->drawText(text_pos, font_color, info, false, font_size);

        ImGui::EndFrame();
		ImGui::Render();
		pUI->renderDrawData(ImGui::GetDrawData());
	}
}

const ImColor fix_color(uint32_t color)
{
	unsigned char a = (int)((color & 0xFF000000) >> 24);
	unsigned char r = (int)((color & 0x00FF0000) >> 16);
	unsigned char g = (int)((color & 0x0000FF00) >> 8);
	unsigned char b = (int)((color & 0x000000FF));

	//FLog("RGBA: 0x%X, 0x%X, 0x%X, 0x%X", r, g, b, a);
	ImColor col = ImColor(r, g, b, a);
	//FLog("color: %f, %f, %f, %f", col.Value.x, col.Value.y, col.Value.z, col.Value.w);
	return col;
}

RwTexture* MaterialTextGenerator::Generate(const char* text, int size, int font_size, bool bold, uint32_t font_color, uint32_t background_color, int alignment)
{
	int width, height;
	GetMaterialSize(size, &width, &height);
    FLog("MaterialTextGenerator::Generate");
	RwRaster* raster = RwRasterCreate(width, height, 32, rwRASTERFORMAT8888 | rwRASTERTYPECAMERATEXTURE);
    FLog("MaterialTextGenerator::Generate1");
    RwTexture* bufferTexture = RwTextureCreate(raster);
    FLog("MaterialTextGenerator::Generate2");

	if (raster && bufferTexture)
	{
        FLog("MaterialTextGenerator::Generate3");
        m_camera->frameBuffer = raster;
        FLog("MaterialTextGenerator::Generate4");
        CVisibilityPlugins::SetRenderWareCamera(m_camera);
        FLog("MaterialTextGenerator::Generate5");
		// RwCameraClear
        RwCameraClear(m_camera, reinterpret_cast<RwRGBA *>(&background_color), 3);
        FLog("MaterialTextGenerator::Generate6");
		RwCameraBeginUpdate((RwCamera*)m_camera);
		DefinedState();
        FLog("MaterialTextGenerator::Generate7");
		Render(text, ImVec2(width, height), font_size, bold, fix_color(font_color), fix_color(background_color), alignment);
        FLog("MaterialTextGenerator::Generate8");
		RwCameraEndUpdate((RwCamera*)m_camera);
	}

	return bufferTexture;
}

void MaterialTextGenerator::GetMaterialSize(int size, int* x, int* y)
{
	int sizes[14][2] = {
			{ 32, 32 } , { 64, 32 }, { 64, 64 }, { 128, 32 }, { 128, 64 }, { 128,128 }, { 256, 32 },
			{ 256, 64 } , { 256, 128 } , { 256, 256 } , { 512, 64 } , { 512,128 } , { 512,256 } , { 512,512 }
	};

	int del = (size / 10) - 1;
	*x = sizes[del][0];
	*y = sizes[del][1];
}

    inline float Clamp(float v, float lo, float hi) { return std::max(lo, std::min(hi, v)); }

    inline bool Near100(float v) { return std::fabs(v - 100.0f) < 0.5f; }

    inline bool IsNeutral(const VehTune& t)
    {
        return Near100(t.speedPct) && Near100(t.accelPct) && Near100(t.inertiaPct) && Near100(t.massPct) &&
               Near100(t.dragPct) && Near100(t.brakePct) && Near100(t.steerPct) && Near100(t.tractionPct) &&
               Near100(t.tracLossPct) && Near100(t.turnPct) && Near100(t.suspForcePct) &&
               Near100(t.suspDampPct) && Near100(t.collisionPct);
    }

    // g_mutex artiq tutulub olmalidir.
    Entry* Find(CVehicleGTA* veh)
    {
        auto it = g_entries.find(veh);
        if (it == g_entries.end()) return nullptr;

        if (it->second.creationTime != veh->m_nCreationTime)
        {
            // Bu adresde artiq basqa masin var (kohne masin silinib).
            // Yeni masin oyunun oz handling-i ile yaranib, bizim kopyaya baxmir.
            delete it->second.copy;
            g_entries.erase(it);
            return nullptr;
        }
        return &it->second;
    }

    Entry& FindOrCreate(CVehicleGTA* veh)
    {
        if (Entry* e = Find(veh)) return *e;
        Entry e;
        e.creationTime = veh->m_nCreationTime;
        return g_entries.emplace(veh, e).first->second;
    }

    void RestoreHandling(CVehicleGTA* veh, Entry& e)
    {
        if (!e.copy) return;
        if (veh->m_pHandlingData == e.copy) veh->m_pHandlingData = e.orig;   // ƏVVƏL pointeri qaytar
        veh->m_fMass = e.baseMass;
        delete e.copy;                                                       // SONRA sil
        e.copy = nullptr;
    }

    void ApplyHandling(CVehicleGTA* veh, Entry& e)
    {
        VehTune& t = e.tune;
        t.speedPct    = Clamp(t.speedPct,    50.0f, 250.0f);
        t.accelPct    = Clamp(t.accelPct,    30.0f, 400.0f);
        t.inertiaPct  = Clamp(t.inertiaPct,  20.0f, 300.0f);
        t.massPct     = Clamp(t.massPct,     30.0f, 300.0f);
        t.dragPct     = Clamp(t.dragPct,     20.0f, 300.0f);
        t.brakePct    = Clamp(t.brakePct,    30.0f, 300.0f);
        t.steerPct    = Clamp(t.steerPct,    50.0f, 150.0f);
        t.tractionPct = Clamp(t.tractionPct, 50.0f, 200.0f);
        t.tracLossPct = Clamp(t.tracLossPct, 20.0f, 200.0f);
        t.turnPct     = Clamp(t.turnPct,     30.0f, 300.0f);
        t.suspForcePct= Clamp(t.suspForcePct,30.0f, 300.0f);
        t.suspDampPct = Clamp(t.suspDampPct, 30.0f, 300.0f);
        t.collisionPct= Clamp(t.collisionPct,10.0f, 300.0f);
        if (t.headlight >= VEHTUNER_COLOR_COUNT) t.headlight = 0;

        if (IsNeutral(t))
        {
            RestoreHandling(veh, e);
            return;
        }

        if (!e.copy)
        {
            if (!veh->m_pHandlingData) return;
            e.orig     = veh->m_pHandlingData;
            e.baseMass = veh->m_fMass;
            e.copy     = new tHandlingData(*e.orig);
        }
        veh->m_pHandlingData = e.copy;      // ferdi kopyaya bagla

        const float kSpd = t.speedPct * 0.01f;
        const float kAcc = t.accelPct * 0.01f;
        const float kIne = t.inertiaPct * 0.01f;
        const float kMas = t.massPct  * 0.01f;
        const float kDrg = t.dragPct  * 0.01f;
        const float kBrk = t.brakePct * 0.01f;
        const float kStr = t.steerPct * 0.01f;
        const float kTrc = t.tractionPct * 0.01f;
        const float kTrl = t.tracLossPct * 0.01f;
        const float kTrn = t.turnPct  * 0.01f;

        // Her zaman ORIGINAL-dan hesablayiriq -> tekrar tetbiq edende yigilmir (idempotent).
        const cTransmission& to = e.orig->m_transmissionData;
        cTransmission&       tc = e.copy->m_transmissionData;

        tc.m_fMaxVelocity           = to.m_fMaxVelocity        * kSpd;
        tc.m_fMaxGearVelocity       = to.m_fMaxGearVelocity    * kSpd;
        tc.m_maxReverseGearVelocity = to.m_maxReverseGearVelocity;
        tc.m_fEngineAcceleration    = to.m_fEngineAcceleration * kAcc;
        tc.m_fEngineInertia         = to.m_fEngineInertia      * kIne;

        // Vites cedveli: her tTransmissionGear-in butun sahələri suretdir -> hamisini miqyasla.
        static_assert(sizeof(tTransmissionGear) % sizeof(float) == 0, "tTransmissionGear float deyil");
        constexpr size_t kFloatsPerGear = sizeof(tTransmissionGear) / sizeof(float);
        for (int i = 0; i < 6; ++i)
        {
            const float* src = reinterpret_cast<const float*>(&to.m_aGears[i]);
            float*       dst = reinterpret_cast<float*>(&tc.m_aGears[i]);
            for (size_t j = 0; j < kFloatsPerGear; ++j) dst[j] = src[j] * kSpd;
        }

        HND_BRAKE(e.copy) = HND_BRAKE(e.orig) * kBrk;
        HND_STEER(e.copy) = HND_STEER(e.orig) * kStr;
        HND_MASS(e.copy)  = HND_MASS(e.orig)  * kMas;
        e.copy->m_fMassRecpr = e.orig->m_fMassRecpr / kMas;            // 1/kutle ayrica saxlanir
        e.copy->m_fTurnMass  = e.orig->m_fTurnMass * kMas * kTrn;      // kutle ile birge + oz slideri
        e.copy->m_fDragMult  = e.orig->m_fDragMult * kDrg;

        e.copy->m_fTractionMultiplier = e.orig->m_fTractionMultiplier * kTrc;
        e.copy->m_fTractionLoss       = e.orig->m_fTractionLoss       * kTrl;

        e.copy->m_fSuspensionForceLevel   = e.orig->m_fSuspensionForceLevel   * (t.suspForcePct * 0.01f);
        e.copy->m_fSuspensionDampingLevel = e.orig->m_fSuspensionDampingLevel * (t.suspDampPct  * 0.01f);
        e.copy->m_fCollisionDamageMultiplier = e.orig->m_fCollisionDamageMultiplier * (t.collisionPct * 0.01f);

        // Fizika kutlesi handling-den AYRI saxlanir -> onu da yenile.
        veh->m_fMass = e.baseMass * kMas;
    }
}

namespace VehTuner
{
    VehTune Get(CVehicleGTA* veh)
    {
        if (!veh) return VehTune{};
        std::lock_guard<std::mutex> lk(g_mutex);
        if (Entry* e = Find(veh)) return e->tune;
        return VehTune{};
    }

    void Set(CVehicleGTA* veh, const VehTune& t)
    {
        if (!veh) return;
        std::lock_guard<std::mutex> lk(g_mutex);
        Entry& e = FindOrCreate(veh);
        e.tune = t;
        ApplyHandling(veh, e);
    }

    void Reset(CVehicleGTA* veh)
    {
        if (!veh) return;
        std::lock_guard<std::mutex> lk(g_mutex);
        if (Entry* e = Find(veh))
        {
            RestoreHandling(veh, *e);
            g_entries.erase(veh);
        }
    }

    bool GetHeadlightRGB(CVehicleGTA* veh, uint8_t& r, uint8_t& g, uint8_t& b)
    {
        if (!veh) return false;
        std::lock_guard<std::mutex> lk(g_mutex);
        if (g_entries.empty()) return false;

        Entry* e = Find(veh);
        if (!e) return false;

        const uint8_t id = e->tune.headlight;
        if (id == 0 || id >= VEHTUNER_COLOR_COUNT) return false;

        r = kPalette[id][0];
        g = kPalette[id][1];
        b = kPalette[id][2];
        return true;
    }

    void GetPaletteColor(int id, uint8_t& r, uint8_t& g, uint8_t& b)
    {
        if (id < 0 || id >= VEHTUNER_COLOR_COUNT) id = 0;
        r = kPalette[id][0];
        g = kPalette[id][1];
        b = kPalette[id][2];
    }
}

struct MMTouch
{
        float    x = -1.0f, y = -1.0f;
        bool     down = false, pressed = false, released = false;
        unsigned events = 0;
};
static MMTouch g_mmTouch;

namespace
{
        struct TouchEv { int type; float x; float y; };

        std::mutex          s_touchMtx;
        std::deque<TouchEv> s_touchQ;

        void PushTouch(int type, float x, float y)
        {
                std::lock_guard<std::mutex> lk(s_touchMtx);

                if (type == TOUCH_MOVE && !s_touchQ.empty() && s_touchQ.back().type == TOUCH_MOVE) {
                        s_touchQ.back().x = x;
                        s_touchQ.back().y = y;
                        return;
                }
                if (s_touchQ.size() > 512) s_touchQ.clear();
                s_touchQ.push_back({ type, x, y });
        }

        void ProcessTouchQueue()
        {
                std::lock_guard<std::mutex> lk(s_touchMtx);

                g_mmTouch.pressed  = false;
                g_mmTouch.released = false;

                while (!s_touchQ.empty())
                {
                        const TouchEv e = s_touchQ.front();
                        s_touchQ.pop_front();

                        g_mmTouch.x = e.x;
                        g_mmTouch.y = e.y;
                        g_mmTouch.events++;

                        if (e.type == TOUCH_PUSH) { g_mmTouch.down = true;  g_mmTouch.pressed  = true; break; }
                        if (e.type == TOUCH_POP)  { g_mmTouch.down = false; g_mmTouch.released = true; break; }
                }
        }
}

        static bool SliderRow(int idx, const Rc& row, const char* label, float* v, float mn, float mx, float dt, bool canInput)
        {
                const int id = ID_SLIDER + idx * 4;
                bool changed = false;

                const float cy = row.y + S(47.0f);
                const Rc minus{ row.x + S(14.0f), cy - S(15.0f), S(30.0f), S(30.0f) };
                const Rc plus { row.x + row.w - S(44.0f), cy - S(15.0f), S(30.0f), S(30.0f) };
                const float tx0 = row.x + S(60.0f), tx1 = row.x + row.w - S(60.0f);
                const Rc hit{ tx0 - S(16.0f), row.y + S(22.0f), (tx1 - tx0) + S(32.0f), row.h - S(22.0f) };

                if (canInput)
                {
                        const MMTouch& t = g_mmTouch;
                        if (t.pressed && s_act == 0) {
                                if (In(minus, t.x, t.y))      s_act = id + 1;
                                else if (In(plus, t.x, t.y))  s_act = id + 2;
                                else if (In(hit, t.x, t.y))   s_act = id;
                        }
                        if (s_act == id && t.down) {
                                const float f  = Clamp((t.x - tx0) / (tx1 - tx0), 0.0f, 1.0f);
                                const float nv = floorf(mn + f * (mx - mn) + 0.5f);
                                if (nv != *v) { *v = nv; changed = true; }
                        }
                        if (t.released) {
                                float nv = *v;
                                if (s_act == id + 1 && In(minus, t.x, t.y)) nv = Clamp(*v - 5.0f, mn, mx);
                                if (s_act == id + 2 && In(plus,  t.x, t.y)) nv = Clamp(*v + 5.0f, mn, mx);
                                if (nv != *v) { *v = nv; changed = true; }
                        }
                }

                const bool dragging = (s_act == id && g_mmTouch.down);
                const float target = Clamp((*v - mn) / (mx - mn), 0.0f, 1.0f);
                s_frac[idx] = Smooth(s_frac[idx], target, dt, dragging ? 40.0f : 14.0f);
                const float fr = s_frac[idx];

        static void Arrow(float cx, float cy, float dir, ImU32 c)
        {
                const float tip = cx + dir * S(18.0f), base = cx + dir * S(2.0f), tail = cx - dir * S(18.0f);

                s_dl->AddRectFilled(ImVec2(x0, cy - S(5.0f)), ImVec2(x1, cy + S(5.0f)), c);
        }

        {
                s_blink = Smooth(s_blink, VehControls::BlinkOn() ? 1.0f : 0.0f, dt, 28.0f);

                static const char* kNames[3] = { "Sol", "Avariya", "Sağ" };
                static const uint8_t kModes[3] = { VEH_SIGNAL_LEFT, VEH_SIGNAL_HAZARD, VEH_SIGNAL_RIGHT };

                for (int k = 0; k < 3; ++k)
                {
                        const Rc b{ blk.x + S(16.0f) + k * (bw + gap), blk.y + S(34.0f), bw, blk.h - S(50.0f) };
                        const int id = ID_SIG + k;
                        if (Tap(id, b, canInput) && veh) VehControls::ToggleSignal(veh, kModes[k]);

                        const float p = Prs(id, dt);
                        const Rc d = Shrink(b, S(1.5f) * p);
                        const bool on = (cur == kModes[k]);
                        const float glow = on ? (0.28f + 0.72f * s_blink) : 0.0f;

                        Fill(d, on ? Orange(30 + (int)(110.0f * glow)) : Col(36, 40, 52, 255), S(9.0f));

                        const float cx = d.x + d.w * 0.5f, cy = d.y + d.h * 0.38f;
                        if (k == 0)      Arrow(cx, cy, -1.0f, ic);
                        else if (k == 2) Arrow(cx, cy, +1.0f, ic);
                        else {
                                s_dl->AddTriangle(ImVec2(cx, cy - S(14.0f)), ImVec2(cx - S(16.0f), cy + S(12.0f)), ImVec2(cx + S(16.0f), cy + S(12.0f)), ic, S(2.2f));

                                Dot(cx, cy + S(7.0f), S(1.7f), ic);
                        }
                        TxtC(S(15.0f), Rc{ d.x, d.y + d.h * 0.62f, d.w, d.h * 0.34f }, on ? Col(255, 224, 180, 255) : Col(190, 196, 212, 255), kNames[k], true);
                }
        }

                const float saveA = s_A;
                s_A = saveA * pe;

                TxtB(S(24.0f), cx0, P.y + S(68.0f), Col(240, 243, 250), kPages[s_page]);
                Txt(S(14.0f), cx0, P.y + S(98.0f), Col(124, 132, 152), kSubs[s_page]);

                struct Spec { const char* label; float* v; float mn; float mx; };

                if (s_page == 0)          // NEZARET
                {
                        const Rc r0{ cx0, P.y + S(124.0f), cw, S(66.0f) };
                        const Rc r1{ cx0, P.y + S(196.0f), cw, S(66.0f) };
                        const Rc r2{ cx0, P.y + S(268.0f), cw, S(66.0f) };

                        if (ToggleRow(0, r0, "Mühərrik", s_eng ? "İşləyir" : "Söndürülüb", s_eng, dt, canInput) && pVehW)
                                pVehW->ApplyEngineState(s_eng ? 0 : 1);
                        if (ToggleRow(1, r1, "Fara", s_lig ? "Yanır" : "Sönülü", s_lig, dt, canInput) && pVehW)
                                pVehW->ApplyLightState(s_lig ? 0 : 1);
                        if (ToggleRow(2, r2, "Qapılar", s_lock ? "Kilidli" : "Açıq", s_lock, dt, canInput) && pVehW)
                                pVehW->SetDoorState(s_lock ? 0 : 1);

                        SignalBlock(Rc{ cx0, P.y + S(346.0f), cw, S(124.0f) }, s_sig, gtaVeh, dt, canInput);
                }
                else if (s_page == 4)     // FAR
                {
                        char hb[40];
                        if (s_tune.headlight == 0) snprintf(hb, sizeof(hb), "Far rəngi: Standart");
                        else                       snprintf(hb, sizeof(hb), "Far rəngi: #%d", (int)s_tune.headlight);
                        Txt(S(17.0f), cx0, P.y + S(128.0f), Col(214, 220, 232), hb);
                        Txt(S(13.0f), cx0, P.y + S(152.0f), Col(124, 132, 152), "Hazırda yalnız hazır palitra — sərbəst rəng seçimi üçün VehTuner/Coronas faylları lazımdır.");

                        for (int i = 0; i < VEHTUNER_COLOR_COUNT; ++i)
                        {
                                uint8_t r, g, b;
                                VehTuner::GetPaletteColor(i, r, g, b);

                                const int   col = i % 8, rowi = i / 8;
                                const float cxs = cx0 + S(24.0f) + col * S(58.0f);
                                const float cys = P.y + S(200.0f) + rowi * S(58.0f);
                                const Rc    sq{ cxs - S(24.0f), cys - S(24.0f), S(48.0f), S(48.0f) };

                                const int sid = ID_SWATCH + std::min(i, 23);
                                if (Tap(sid, sq, canInput)) { s_tune.headlight = (uint8_t)i; changed = true; }

                                const float p = Prs(sid, dt);
                                Dot(cxs, cys, S(20.0f) - S(2.0f) * p, Col(r, g, b, 255));
                                if (s_tune.headlight == i) s_dl->AddCircle(ImVec2(cxs, cys), S(25.0f), Col(255, 255, 255, 230), 28, S(2.2f));
                        }
                }
                else if (s_page == 5)    // KOMANDALAR
                {
                        struct CmdInfo { const char* cmd; const char* desc; };
                        static const CmdInfo kCmds[] = {
                                { "/mm",      "Maşın menyusunu aç/bağla" },
                                { "/sinyal",  "sol | sag | avariya | off" },
                                { "/far N",   "Far rəngi (0-15)" },
                                { "/suret N", "Maksimum sürət" },
                                { "/manevr N","Manevr bucağı" },
                                { "/tormoz N","Tormoz gücü" },
                                { "/gizle",   "Bütün UI-ı gizlət/geri qaytar" },
                                { "/yt",      "YouTube paneli" },
                        };
                        const int n = (int)(sizeof(kCmds) / sizeof(kCmds[0]));
                        for (int i = 0; i < n; ++i)
                        {
                                const Rc row{ cx0, P.y + S(124.0f + 52.0f * i), cw, S(44.0f) };
                                Fill(row, Col(27, 30, 40, 255), S(9.0f));
                                Txt(S(15.0f), row.x + S(16.0f), row.y + (row.h - S(15.0f)) * 0.5f, Accent(0.0f), kCmds[i].cmd);
                                Txt(S(14.0f), row.x + S(160.0f), row.y + (row.h - S(14.0f)) * 0.5f, Col(200, 204, 216), kCmds[i].desc);
                        }
                }
                else                      // HANDLING SEHIFELERI
                {
                        Spec sp[5];
                        int  n = 0;
                        if (s_page == 1) {
                                sp[n++] = { "Yerindən götürmə (təcil)",    &s_tune.accelPct,   30.0f, 400.0f };
                                sp[n++] = { "Maksimum sürət",              &s_tune.speedPct,   50.0f, 250.0f };
                                sp[n++] = { "Mühərrik ətaləti (az = tez)", &s_tune.inertiaPct, 20.0f, 300.0f };
                                sp[n++] = { "Çəki / kütlə",                &s_tune.massPct,    30.0f, 300.0f };
                                sp[n++] = { "Hava müqaviməti",             &s_tune.dragPct,    20.0f, 300.0f };
                        } else if (s_page == 2) {
                                sp[n++] = { "Tormoz gücü",                 &s_tune.brakePct,    30.0f, 300.0f };
                                sp[n++] = { "Manevr bucağı",               &s_tune.steerPct,    50.0f, 150.0f };
                                sp[n++] = { "Yol tutuş (traction)",        &s_tune.tractionPct, 50.0f, 200.0f };
                                sp[n++] = { "Sürüşmə itkisi (az = yaxşı)", &s_tune.tracLossPct, 20.0f, 200.0f };
                                sp[n++] = { "Fırlanma ətaləti",            &s_tune.turnPct,     30.0f, 300.0f };
                        } else {
                                sp[n++] = { "Asma gücü",                   &s_tune.suspForcePct, 30.0f, 300.0f };
                                sp[n++] = { "Asma amortizatorları",        &s_tune.suspDampPct,  30.0f, 300.0f };
                                sp[n++] = { "Toqquşma zərəri",             &s_tune.collisionPct, 10.0f, 300.0f };
                        }

                        for (int i = 0; i < n; ++i) {
                                const Rc row{ cx0, P.y + S(124.0f + 72.0f * i), cw, S(66.0f) };
                                if (SliderRow(i, row, sp[i].label, sp[i].v, sp[i].mn, sp[i].mx, dt, canInput)) changed = true;
                        }
                }

                // --- alt sira: Sifirla (yalniz handling sehifelerinde) + status ---
                if (s_page >= 1 && s_page <= 3)
                {
                        const Rc rb{ cx0, P.y + P.h - S(60.0f), S(150.0f), S(42.0f) };
                        if (Tap(ID_RESET, rb, canInput) && gtaVeh) {
                                VehTuner::Reset(gtaVeh);
                                s_tune = VehTune();
                                changed = false;
                                s_flash = 1.0f;
                                for (int k = 0; k < 8; ++k) s_frac[k] = 0.0f;
                                if (pUI && pUI->chat()) pUI->chat()->addDebugMessage("{00FF00}[MM]: Standart dəyərlər qaytarıldı.");
                        }
                        const float rp = Prs(ID_RESET, dt);
                        const Rc rd = Shrink(rb, S(1.5f) * rp);
                        Fill(rd, Accent(0.0f, 20 + (int)(36.0f * rp)), S(9.0f));
                        Stroke(rd, Accent(0.0f, 170), S(9.0f), S(1.3f));
                        TxtC(S(16.0f), rd, Col(232, 238, 250), "Sıfırla", true);

                        if (s_flash > 0.01f) {
                                Dot(cx0 + S(174.0f), rb.y + rb.h * 0.5f, S(4.5f), Col(90, 205, 140, (int)(255.0f * s_flash)));
                                Txt(S(15.0f), cx0 + S(188.0f), rb.y + (rb.h - S(15.0f)) * 0.5f, Col(120, 215, 160, (int)(255.0f * s_flash)), "Tətbiq edildi");
                        }
                }

                s_A = saveA;
        }

        // Her deyisiklik canli tetbiq olunur (original-dan hesablandigi ucun yigilmir)
        if (changed && want && gtaVeh) {
                VehTuner::Set(gtaVeh, s_tune);
                s_flash = 1.0f;
        }

        if (tch.released || !tch.down) s_act = 0;

#if HND_DEBUG
        // Diaqnostika: qirmizi halqa = oyunun hesab etdiyi barmaq yeri
        {
                ImDrawList* d = ImGui::GetForegroundDrawList();
                if (tch.x >= 0.0f) {
                        d->AddCircle(ImVec2(tch.x, tch.y), 22.0f, tch.down ? IM_COL32(255, 60, 60, 255) : IM_COL32(255, 220, 60, 160), 24, 3.0f);
                }
                char dbg[96];
                snprintf(dbg, sizeof(dbg), "T %.0f,%.0f d%d ev%u disp %.0fx%.0f act%d", tch.x, tch.y, (int)tch.down, tch.events, disp.x, disp.y, s_act);
                d->AddText(ImGui::GetFont(), 18.0f, ImVec2(11.0f, disp.y - 29.0f), IM_COL32(0, 0, 0, 255), dbg);
                d->AddText(ImGui::GetFont(), 18.0f, ImVec2(10.0f, disp.y - 30.0f), IM_COL32(255, 255, 255, 255), dbg);
        }
#endif
}
