102: void ImGuiRenderer::drawText(const ImVec2& pos, const ImColor& color, const std::string& text, bool outlined, float font_size)
103: {
104: 	if (text.empty()) return;
105: 
106: 	float sz_font = font_size == 0.0f ? m_font->FontSize : font_size;
107: 
108: 	const char* text_start = text.c_str();
109: 	const char* text_cur = text.c_str();
110: 	const char* text_end = text.c_str() + text.length();
111: 
112: 	ImVec2 pos_cur = pos;
113: 	ImColor color_cur = color;
114: 
115: 	while (*text_cur)
116: 	{
117: 		if (*text_cur == '{' && ((&text_cur[7] < text_end) && text_cur[7] == '}'))
118: 		{
119: 			// print accumulated text
120: 			if (text_cur != text_start)
121: 			{
122: 				drawText(pos_cur, color_cur, text_start, text_cur, outlined, sz_font);
123: 				ImVec2 sz = calculateTextSize(text_start, text_cur, sz_font);
124: 				pos_cur.x += sz.x;
125: 			}
126: 
127: 			// new colorcode
128: 			ImVec4 col;
129: 			if (processInlineHexColor(text_cur + 1, text_cur + 7, col)) {
130: 				color_cur = col;
131: 			}
132: 
133: 			text_cur += 7;
134: 			text_start = text_cur + 1;
135: 		}
136: 		else if (*text_cur == '\n')
137: 		{
138: 			// print accumulated text
139: 			if (text_cur != text_start)
140: 			{
141: 				drawText(pos_cur, color_cur, text_start, text_cur, outlined, sz_font);
142: 			}
143: 
144: 			pos_cur.x = pos.x;
145: 			pos_cur.y += sz_font;
146: 			text_start = text_cur + 1;
147: 		}
148: 		else if (*text_cur == '\t')
149: 		{
150: 			// print accumulated text
151: 			if (text_cur != text_start)
152: 			{
153: 				drawText(pos_cur, color_cur, text_start, text_cur, outlined, sz_font);
154: 				ImVec2 sz = calculateTextSize(text_start, text_cur, sz_font);
155: 				pos_cur.x += sz.x;
156: 			}
157: 
158: 			pos_cur.x += sz_font;
159: 			text_start = text_cur + 1;
160: 		}
161: 
162: 		++text_cur;
163: 	}
164: 
165: 	if (text_cur != text_start) {
166: 		drawText(pos_cur, color_cur, text_start, text_cur, outlined, sz_font);
167: 	}
168: }
